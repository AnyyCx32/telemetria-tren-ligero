    <script>
        // Inicializar mapa
        const map = L.map("map").setView([19.4326, -99.1332], 13);
        L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
            attribution: '© OpenStreetMap contributors'
        }).addTo(map);

        const colores = ["#1f77b4", "#2ca02c", "#ff7f0e", "#9467bd"]; // Colores más profesionales
        const marcadores = {};
        const rutas = {};

        function colorTren(id) { return colores[id % colores.length]; }

        function mostrarError(mensaje) {
            const errorDiv = document.getElementById('errorGlobal');
            errorDiv.style.display = 'block';
            errorDiv.textContent = mensaje;
            setTimeout(() => { errorDiv.style.display = 'none'; }, 5000);
        }

        async function actualizar() {
            const loading = document.getElementById('loadingEstado');
            loading.style.display = 'block';
            try {
                const res = await fetch("/api/estado");
                if (!res.ok) throw new Error(`Error HTTP ${res.status}`);
                const trenes = await res.json();

                const tbody = document.getElementById("tabla");
                tbody.innerHTML = "";

                trenes.forEach(t => {
                    const p = t.posicion;
                    const estado = p.gps_off_s < 35 ? "VERDE" : p.gps_off_s <= 60 ? "AMARILLO" : "ROJO";
                    const estadoClass = estado.toLowerCase();

                    tbody.innerHTML += `
                        <tr>
                            <td><strong>${t.id_tren}</strong></td>
                            <td>${p.lat.toFixed(6)}</td>
                            <td>${p.lon.toFixed(6)}</td>
                            <td>${t.velocidad ? Number(t.velocidad).toFixed(2) : '-'}</td>
                            <td>${t.direccion_txt}</td>
                            <td><span class="${estadoClass}">${estado}</span></td>
                        </tr>
                    `;

                    const punto = [p.lat, p.lon];
                    if (!marcadores[t.id_tren]) {
                        marcadores[t.id_tren] = L.circleMarker(punto, {
                            radius: 8,
                            color: colorTren(t.id_tren),
                            fillColor: colorTren(t.id_tren),
                            fillOpacity: 0.8,
                            weight: 2
                        }).addTo(map);
                    } else {
                        marcadores[t.id_tren].setLatLng(punto);
                    }

                    if (t.historial && t.historial.length >= 2) {
                        const pts = t.historial.map(h => [h.lat, h.lon]);
                        if (!rutas[t.id_tren]) {
                            rutas[t.id_tren] = L.polyline(pts, { 
                                color: colorTren(t.id_tren),
                                weight: 3,
                                opacity: 0.7
                            }).addTo(map);
                        } else {
                            rutas[t.id_tren].setLatLngs(pts);
                        }
                    }
                });
                loading.style.display = 'none';
            } catch (error) {
                loading.style.display = 'none';
                mostrarError('Error al cargar estado: ' + error.message);
            }
        }

        async function actualizarHistorial() {
            const loading = document.getElementById('loadingHistorial');
            loading.style.display = 'block';
            try {
                const res = await fetch("/api/tabla");
                if (!res.ok) throw new Error(`Error HTTP ${res.status}`);
                const datos = await res.json();

                const tbody = document.getElementById("tablaHistorial");
                tbody.innerHTML = "";

                datos.forEach(d => {
                    const estado = d.gps_off_s < 35 ? "VERDE" : d.gps_off_s <= 60 ? "AMARILLO" : "ROJO";
                    const estadoClass = estado.toLowerCase();
                    const fecha = new Date(d.timestamp);
                    const horaLocal = fecha.toLocaleString('es-MX', { hour12: false });

                    tbody.innerHTML += `
                        <tr>
                            <td><strong>${d.id_tren}</strong></td>
                            <td>${d.lat.toFixed(5)}</td>
                            <td>${d.lon.toFixed(5)}</td>
                            <td><span class="${estadoClass}">${estado}</span></td>
                            <td>${horaLocal}</td>
                        </tr>
                    `;
                });
                loading.style.display = 'none';
            } catch (error) {
                loading.style.display = 'none';
                mostrarError('Error al cargar historial: ' + error.message);
            }
        }

        // Inicializar y actualizar periódicamente
        actualizar();
        actualizarHistorial();
        setInterval(() => {
            actualizar();
            actualizarHistorial();
        }, 15000);
    </script>
